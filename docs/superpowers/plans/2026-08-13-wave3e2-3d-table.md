## 設計上の決めごと（読まずに実装しないこと）

### 置き場所を出す部分を純粋関数にする

WebGL は自動試験にかけられない。だから**「盤面 → どの牌をどこへ置くか」を Three.js に触らない関数**として切り出し、そこを試験する。`table.ts` は受け取った配置を描くだけにする。

こうすると、牌が変な場所に出る不具合の大半が、描画を動かさずに捕まる。

### 牌の面は SVG から焼く

`atlas.ts` の `drawPlaceholderAtlas` は「1m」のような文字を書くだけの仮のものである。**Wave 3e-1 で描いた SVG をそのままアトラスへ焼く。**見た目の定義が 2 か所にあると必ずずれる。

SVG を `Image` 経由で canvas へ描くのは非同期なので、**アトラスができるまでは仮のもので描き、できたら差し替える**。テクスチャの `needsUpdate` を立てれば済む。

### 自席から見た向きに直してから置く

`layout.ts` の関数は**相対席**（自分が 0）を取る。盤面が持つのは**絶対席**なので、`relativeSeat(absolute, you)` を通してから渡す。**ここを取り違えると、自分の手牌が対面に並ぶ。**

### 山は本物の枚数を持たない

サーバは山の中身を送らない（送ったら不正になる）。**残り枚数だけが分かる。**そこで山は「残り枚数ぶんの伏せ牌」を四辺へ並べる見せかけにする。**牌の中身は持たない。**

### 操作は HTML のまま重ねる

鳴きのボタン・持ち時間・点数・和了の知らせは、3e-1 の DOM をそのまま卓の上に重ねる。3D の中にボタンを作らない。**押しやすさと読みやすさで HTML が勝るうえ、既に試験がある。**

```typescript
import type { GameState } from "../game/state";
import type { Seat } from "../protocol/Seat";
import type { Tile } from "../protocol/Tile";
import {
  TILE,
  discardPosition,
  handPosition,
  relativeSeat,
  seatRotation,
  wallPosition,
  type Vec3,
} from "./layout";

/** 卓のどこに置かれる牌か。 */
export type PlacementKind = "hand" | "drawn" | "river" | "meld" | "wall" | "dora";

export type Placement = {
  /** 牌1枚を指す鍵。**重複すると牌が消えたり重なったりする。** */
  key: string;
  kind: PlacementKind;
  /** 絶対席。見せる向きは `position` に織り込み済み。 */
  seat: Seat;
  /** 伏せ牌は 0 を入れる。`faceUp` が偽なら中身は使わない。 */
  encoded: Tile;
  position: Vec3;
  rotationY: number;
  faceUp: boolean;
};

/** 河で横に倒す角度。 */
const SIDEWAYS = Math.PI / 2;

/**
 * 盤面から牌の置き場所を出す。
 *
 * **Three.js に触らない。**WebGL は自動試験にかけられないので、
 * 「どの牌をどこへ置くか」だけをここで決めて試験する。
 */
export function placementsFor(state: GameState, viewer: Seat = state.you): Placement[] {
  const out: Placement[] = [];
  if (state.round === null) {
    return out;
  }

  for (let absolute = 0; absolute < 4; absolute += 1) {
    const seat = state.seats[absolute];
    if (!seat) {
      continue;
    }
    const relative = relativeSeat(absolute, viewer);
    const facing = seatRotation(relative);
    const mine = absolute === state.you;

    // 手牌。自分だけ表、他家は伏せる。
    const hand = mine ? state.hand : [];
    const size = mine ? hand.length : seat.handSize;
    for (let index = 0; index < size; index += 1) {
      out.push({
        key: `hand-${absolute}-${index}`,
        kind: "hand",
        seat: absolute,
        encoded: hand[index] ?? 0,
        position: handPosition(relative, index, size + (mine && state.drawn !== null ? 1 : 0)),
        rotationY: facing,
        faceUp: mine,
      });
    }

    // ツモ牌は手牌から離す。
    if (mine && state.drawn !== null) {
      const spot = handPosition(relative, size + 1, size + 2);
      out.push({
        key: `drawn-${absolute}`,
        kind: "drawn",
        seat: absolute,
        encoded: state.drawn,
        position: spot,
        rotationY: facing,
        faceUp: true,
      });
    }

    // 河。
    seat.river.forEach((discarded, index) => {
      out.push({
        key: `river-${absolute}-${index}`,
        kind: "river",
        seat: absolute,
        encoded: discarded.tile,
        position: discardPosition(relative, index),
        rotationY: facing + (discarded.riichi ? SIDEWAYS : 0),
        faceUp: true,
      });
    });

    // 副露。手牌の右外へ並べる。
    let meldColumn = 0;
    seat.melds.forEach((meld, meldIndex) => {
      meld.tiles.forEach((tile, tileIndex) => {
        // **暗槓の中身は見せない。**両端だけ伏せるのが本来だが、
        // ここでは4枚とも伏せる。
        const concealed = meld.kind === "ankan";
        const spot = handPosition(relative, meldColumn, 1);
        out.push({
          key: `meld-${absolute}-${meldIndex}-${tileIndex}`,
          kind: "meld",
          seat: absolute,
          encoded: concealed ? 0 : tile,
          position: { x: spot.x + 6.5 * Math.cos(facing), y: spot.y, z: spot.z - 6.5 * Math.sin(facing) },
          rotationY: facing,
          faceUp: !concealed,
        });
        meldColumn += 1;
      });
      meldColumn += 0.4;
    });
  }

  // 山。**中身はサーバから来ない。**残り枚数ぶんの伏せ牌を並べる。
  for (let index = 0; index < state.wallRemaining; index += 1) {
    const side = index % 4;
    out.push({
      key: `wall-${index}`,
      kind: "wall",
      seat: side as Seat,
      encoded: 0,
      position: wallPosition(side, Math.floor(index / 4)),
      rotationY: seatRotation(side),
      faceUp: false,
    });
  }

  // ドラ表示。卓の中央へ横一列。
  state.doraIndicators.forEach((indicator, index) => {
    out.push({
      key: `dora-${index}`,
      kind: "dora",
      seat: 0,
      encoded: indicator,
      position: { x: (index - 2) * TILE.width, y: TILE.depth / 2, z: 0 },
      rotationY: 0,
      faceUp: true,
    });
  });

  return out;
}
```

`apps/web/src/scene/face-atlas.ts`:

```typescript
import { tileBackSvg, tileFaceSvg } from "../ui/tile-face";
import { ATLAS, BACK_INDEX, BODY_INDEX } from "./atlas";

/**
 * アトラスの1セルを SVG のデータ URL にする。
 *
 * **牌の面は `ui/tile-face.ts` が唯一の定義である。**ここで描き直すと
 * 2D と 3D で見た目がずれる。
 */
export function faceDataUrl(cell: number): string {
  if (!Number.isInteger(cell) || cell < 0 || cell >= ATLAS.cells) {
    throw new Error(`アトラスのセルが範囲外: ${cell}`);
  }
  const svg =
    cell === BACK_INDEX
      ? tileBackSvg()
      : cell === BODY_INDEX
        ? bodySvg()
        : tileFaceSvg(cell);
  return `data:image/svg+xml;charset=utf-8,${encodeURIComponent(svg)}`;
}

/** 牌の側面と底。象牙色の無地。 */
function bodySvg(): string {
  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 40 54"><rect width="40" height="54" fill="#f0e6c8"/></svg>`;
}

/** 画像を読む手立て。試験では差し替える。 */
export type LoadImage = (url: string) => Promise<CanvasImageSource>;

/** ブラウザで画像を読む。 */
export const loadViaImage: LoadImage = (url) =>
  new Promise((resolve, reject) => {
    const image = new Image();
    image.onload = () => resolve(image);
    image.onerror = () => reject(new Error("牌面を読めない"));
    image.src = url;
  });

/**
 * アトラスへ39セルを描く。
 *
 * **非同期である。**できるまでは `atlas.ts` の仮のもので描き、
 * できたらテクスチャの `needsUpdate` を立てて差し替える。
 */
export async function paintAtlas(
  canvas: HTMLCanvasElement,
  load: LoadImage = loadViaImage,
): Promise<void> {
  const context = canvas.getContext("2d");
  if (context === null) {
    throw new Error("2D コンテキストを取得できない");
  }
  const cellW = canvas.width / ATLAS.columns;
  const cellH = canvas.height / ATLAS.rows;

  const images = await Promise.all(
    Array.from({ length: ATLAS.cells }, (_, cell) => load(faceDataUrl(cell))),
  );
  images.forEach((image, cell) => {

  });
}
```

## Self-Review

**仕様の網羅:** 仕様 3.1 の「3D卓 + 2Dキャラ演出」のうち 3D 卓を満たす。2D キャラと演出のタイムラインは次のウェーブ。

**このウェーブがやらないこと:** 2D キャラ、音、演出のタイムライン（`timeline/` は完成済みだが結線しない）、カメラの移動、牌を動かす補間。**まず静止した正しい盤面を出す。**動きはそのあとで足せる。

**認められた妥協:**

- 牌の面は SVG から焼いたもので、本物の牌画像ではない。差し替えは `assets/` のマニフェストへ移すときに行う。
- 山は見せかけである。サーバは中身を送らないので、残り枚数ぶんの伏せ牌を並べるだけ。
- 暗槓は4枚とも伏せる。本来は両端2枚だけ伏せる。
- 牌は瞬間移動する。補間は演出のウェーブで足す。

**型の整合:** `Placement` は Task 1 で定義し、Task 3・4 はそれを読むだけ。`GameState` と `Command` は Wave 3e-1 のものをそのまま使う。

**確認済みの事実:** `layout.ts` の座標系（卓の中心が原点、卓面が y=0、自席から見て奥が -z、単位は牌の幅）、`ATLAS` が 8列×5行39セルで `atlasIndexOf` が 0..=36 をそのままセル番号にすること、`applyFaceUv(geometry, encoded, faceUp)` が牌面の UV だけを動かすことは、`apps/web/src/scene/` の実コードから読み取った。
